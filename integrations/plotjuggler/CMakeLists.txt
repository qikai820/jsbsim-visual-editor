cmake_minimum_required(VERSION 3.16)
project(JSBSimPlotJuggler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 REQUIRED COMPONENTS Widgets Network Xml)
# This integration does not use PlotJuggler's WebAssembly parser. The pinned
# upstream build treats an existing target as an opt-out for downloading the
# 130 MB Wasmer runtime; wasmer_FOUND stays false, so no Wasmer code is built.
add_library(wasmer::wasmer INTERFACE IMPORTED GLOBAL)
add_subdirectory("${PLOTJUGGLER_SOURCE}" upstream)
# Upstream CSV assumes it is the top-level project for this include path.
target_include_directories(csv_parser_lib PUBLIC "${PLOTJUGGLER_SOURCE}/3rdparty/date-3.0.4/include")
add_library(JSBSimLive SHARED jsbsim_live.cpp)
target_link_libraries(JSBSimLive PRIVATE plotjuggler_base Qt5::Widgets Qt5::Network Qt5::Xml)
target_compile_definitions(JSBSimLive PRIVATE QT_PLUGIN)
set_target_properties(JSBSimLive PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(JSBSimMcap SHARED jsbsim_mcap.cpp)
target_include_directories(JSBSimMcap PRIVATE "${PLOTJUGGLER_SOURCE}/plotjuggler_plugins/DataLoadMCAP/3rdparty")
target_link_libraries(JSBSimMcap PRIVATE plotjuggler_base Qt5::Widgets Qt5::Xml LZ4::lz4_static zstd::libzstd_static)
target_compile_definitions(JSBSimMcap PRIVATE QT_PLUGIN)
set_target_properties(JSBSimMcap PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(JSBSimMcapCheck mcap_check.cpp)
target_include_directories(JSBSimMcapCheck PRIVATE "${PLOTJUGGLER_SOURCE}/plotjuggler_plugins/DataLoadMCAP/3rdparty")
target_link_libraries(JSBSimMcapCheck PRIVATE LZ4::lz4_static zstd::libzstd_static)
set_target_properties(JSBSimMcapCheck PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
